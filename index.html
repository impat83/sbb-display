<script>


function updateClock(){

    const now = new Date();

    const hh = String(now.getHours()).padStart(2,"0");

    const mm = String(now.getMinutes()).padStart(2,"0");

    document.getElementById("clock").innerText = hh + ":" + mm;

}


setInterval(updateClock,1000);

updateClock();


async function loadDepartures(){


const response = await fetch("https://transport.opendata.ch/v1/stationboard?station=Alpnach%20Dorf&limit=8");

const data = await response.json();


const table = document.getElementById("board");

table.innerHTML="";


data.stationboard.forEach(train => {


const time = new Date(train.stop.departure);

const hh = String(time.getHours()).padStart(2,"0");

const mm = String(time.getMinutes()).padStart(2,"0");


const delay = train.stop.delay ? "+" + train.stop.delay : "";


const row = `

<tr>

<td class="time">${hh}:${mm}</td>

<td class="line">${train.category}</td>

<td>${train.to}</td>

<td class="delay">${delay}</td>

<td class="platform">${train.stop.platform || ""}</td>

</tr>

`;


table.innerHTML += row;


});


}


loadDepartures();

setInterval(loadDepartures,30000);


</script>
